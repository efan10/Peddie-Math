    = \boxed{ \frac{1}{15}(1+x^2)^\frac{3}{2}(3x^2-2)+C}
\end{gather*}

\problem{9}

\subproblem{a}
\begin{gather*}
    f(x)-f(a)=\int_a^x f'(t)\,dt\\
    \int_a^x f'(t)\,dt =
    \begin{bmatrix}
        u=f'(t) & dv = dt\\
        du=f''(t) \,dt& v=t-x
    \end{bmatrix}
    (t-x)f'(t)\Big|_a^x - \int_a^x (t-x)f''(t)\,dt\\
    f(x)-f(a)=(t-x)f'(t)\Big|_a^x - \int_a^x (t-x)f''(t)\,dt\\
    f(x)=f(a)+(x-a)f'(a) + \int_a^x (x-t)f''(t)\, dt\\
    \boxed{f(x)=L(x)+\int_a^x (x-t)f''(t)\, dt}
\end{gather*}

\subproblem{b}
\begin{gather*}
    |f(x)-L(x)|=|\int_a^x (x-t)f''(t)\, dt|\\
    |f''(t)| \le M \implies \int_a^x (x-t)f''(t)\, dt \le M \int_a^x x-t\, dt\\
    M \int_a^x x-t\, dt= -\frac{M(x-t)^2}{2}\Big|^x_a = \frac{M}{2}(x-a)^2\\
    \boxed{|f(x)-L(x)| \le \frac{M}{2}(x-a)^2}
\end{gather*}

\subproblem{c}
\begin{gather*}
    f(x)-L(x)=\int_a^x (x-t)f''(t)\,dt\\
    f''(t)\ge  0 \implies \int_a^x (x-t)f''(t)\,dt\ge 0\\
    \boxed{f(x) \ge L(x)}
\end{gather*}

\subproblem{d}
\begin{gather*}
    \int_a^x (x-t)f''(t)\, dt =
    \begin{bmatrix}
        u=f''(t) & dv=(x-t)\, dt\\
        du = f'''(t)\, dt & v = -\frac{(x-t)^2}{2}
    \end{bmatrix}\\
    -\frac{1}{2}(x-t)^2f''(t)\Big|^x_a + \int_a^x \frac{1}{2}(x-t)^2f'''(t)\, dt\\
    \boxed{f(x)=f(a)+f'(a)(x-a)+\frac{f''(a)}{2}(x-a)^2+\int_a^x \frac{f'''(t)}{2}(x-t)^2 \, dt}
\end{gather*}

\problem{10}

\subproblem{a}
\begin{gather*}
    B(m,n) = \int_0^1 x^m (1-x)^n \, dx\\
    =\begin{bmatrix}
        u=(1-x)^n & dv=x^m\, dx\\
        du=-n(1-x)^{n-1} & v=\frac{x^{m+1}}{m+1}
    \end{bmatrix}
    \frac{1}{m+1}(1-x)^nx^{m+1}\Big|^1_0 \\+  \int_0^1 \frac{n}{m+1}x^{m+1}(1-x)^{n-1}\\
    = \boxed{\frac{n}{m+1} B(m+1,n-1)}
\end{gather*}

\subproblem{b}
\begin{align*}
    B(m,n)&=\frac{n}{m+1}B(m+1,n-1)\\
    &=\frac{n}{m+1} \cdot \frac{n-1}{m+2}B(m+2,n-2)\\
    &= \frac{n}{m+1} \cdot \frac{n-1}{m+2} \cdot \frac{n-2}{m+3}B(m+3,n-3)\\
    &\dots \\
    &=\frac{n(n-1) \cdots 1}{(m+1)(m+2)\cdots (m+n)}B(m+n,0)\\
    &=\frac{n!}{(m+1)(m+2)\cdots (m+n)(m+n+1)}\\
    &=\boxed{\frac{m!\, n!}{(m+n+1)!}}
\end{align*}

\subproblem{c}
\begin{gather*}
    \int_0^1 \frac{x^4(1-x)^4}{1+x^2} \, dx = \int_0^1 \left( x^6 - 4x^5 + 5x^4 - 4x^2 + 4 - \frac{4}{1+x^2} \right) dx\\
    =\frac{x^7}{7} - \frac{2x^6}{3} + x^5 - \frac{4x^3}{3} + 4x - 4\arctan(x)\Big|^1_0=\boxed{\frac{22}{7}-\pi}\\
    \int_0^1 \frac{x^4(1-x)^4}{1+x^2} \, dx > 0 \implies \boxed{\pi < \frac{22}{7}}
\end{gather*}

\subproblem{d}
\begin{gather*}
    B(4,4)=\int_0^1 x^4(1-x)^4 \, dx = \frac{4!\cdot 4!}{9!} = \frac{1}{630}\\
    \frac{1}{2}\le \frac{1}{1+x^2} \le 1  \\
