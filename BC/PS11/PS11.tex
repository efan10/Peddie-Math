\problem{1}

\subproblem{c}
\begin{gather*}
    u=\ln x\\
    du= \frac{1}{x}\, dx\\
    \int \frac{\ln^2x}{x}\, dx = \int u^2\, du = \frac{u^3}{3}+C=\boxed{\frac{\ln^3x}{3}+C}
\end{gather*}

\subproblem{d}
Both are correct.

\problem{2}

\subproblem{e}
\begin{gather*}
    u=\frac{3x^2}{\sqrt{2}}\\
    du=3\sqrt{2}x\, dx\\
    \int \frac{x}{2+9x^4}\, dx=\frac{1}{6\sqrt{2}}\int \frac{1}{1+u^2}\, du\\ = \frac{1}{6\sqrt{2}} \arctan u\,+C= \boxed{\frac{1}{6\sqrt{2}} \arctan \frac{3x^2}{\sqrt{2}}+C}
\end{gather*}

\subproblem{g}
\begin{gather*}
    u=\frac{1}{3}\tan 4x\\
    du=\frac{4}{3}\sec^24x\, dx\\
    \int \frac{\sec^2 4x}{9+\tan^24x}\, dx=\frac{1}{12}\int \frac{1}{1+u^2} \, du \\= \frac{1}{12}\arctan u+C =\boxed{\frac{1}{12}\arctan \frac{\tan 4x}{3}+C}
\end{gather*}

\problem{3}

\subproblem{b}
\begin{gather*}
    u=x+1\\
    du=dx\\
    \int x\cbrt{x+1}\, dx = \int (u-1)\cbrt{u} \, du = \int u^\frac{4}{3}-u^\frac{1}{3}\, du \\= \boxed{\frac{3}{7}(x+1)^\frac{7}{3}+\frac{3}{4}(x+1)^\frac{4}{3}+C}
\end{gather*}

\problem{4}

\subproblem{c}
\begin{gather*}
    \int_0^\frac{1}{4} \sin^2\pi x \, dx= \int_0^\frac{1}{4} \frac{1-\cos 2\pi x}{2}\, dx= \boxed{\frac{1}{8} -\frac{1}{4\pi}}
\end{gather*}

\subproblem{d}
\begin{gather*}
    u=\cos x\\
    du=-\sin x \, dx\\
    \int \sin^3x\, dx = \int (1-\cos ^2x)\sin x\, dx\\
    =\int u^2 -1 \, du = \frac{u^3}{3}-u+C=\boxed{\frac{\cos^3x}{3}-\cos x+C}
\end{gather*}

\subproblem{f}
\begin{gather*}
    u=\frac{\pi}{2}-x\\
    du =-dx\\
    \int_0^\frac{\pi}{2}\sin^2x \, dx = \int_0^\frac{\pi}{2} \sin^2 (\frac{\pi}{2}-x)\, dx = \int_0^\frac{\pi}{2}\cos ^2x\, dx
\end{gather*}

\problem{6}

\subproblem{a}
\begin{gather*}
    u=4-x\\
    du=-dx\\
    \int_0^3 \frac{1}{\sqrt{4-x}}\, dx= \int_1^4 \frac{1}{\sqrt{u}}\, du= 2\sqrt{u} \Big|^4_1=\boxed{2}
\end{gather*}

\subproblem{b}
\begin{gather*}
    u=\frac{x}{2}\\
    du=\frac{dx}{2}\\
    \int_0^1 \frac{1}{\sqrt{4-x^2}}\,dx = \int_0^\frac{1}{2} \frac{1}{\sqrt{1-u^2}} \, du= \arcsin \frac{1}{2}=\boxed{\frac{\pi}{6}}
\end{gather*}

\subproblem{c}
\begin{gather*}\\
    u=\frac{x^2}{3}\\
    du = \frac{2}{3}dx\\
    \int_0^{\sqrt{3}} \frac{x}{9+x^4}\, dx = \frac{1}{6}\int_0^1 \frac{1}{1+u^2} \, du = \frac{1}{6} \arctan 1=\boxed{\frac{\pi}{24}}
\end{gather*}

\subproblem{d}
\begin{gather*}
    u= 9+x^2\\
    du=2x\, dx\\
    \int_0^2 \frac{2x}{9+x^2}\, dx= \int_9^{13} \frac{1}{u}\, du= \boxed{\ln \frac{13}{9}}
\end{gather*}

\subproblem{e}
\begin{gather*}
    u=\arctan 2x\\
    du=\frac{2}{1+4x^2}\, dx\\
    \int_0^\frac{1}{2}\frac{\arctan 2x}{1+4x^2} \, dx= \int_0^\frac{\pi}{4} \frac{u}{2}\, du = \boxed{\frac{\pi^2}{64}}
\end{gather*}

\subproblem{f}
\begin{gather*}
    u=\sqrt x\\
    du =\frac{1}{2\sqrt{x}}\, du\\
    \int_0^\frac{\pi^2}{16} \frac{\tan \sqrt{x}}{\sqrt{x}}=\int_0^\frac{\pi}{4} 2\tan u\, du=\int_0^\frac{\pi}{4} \frac{2\sin u}{\cos u}\, du\\
    t=\cos u\\
    dt=-\sin u\, du\\
    \int_0^\frac{\pi}{4} \frac{2\sin u}{\cos u}\, du= \int_\frac{\sqrt{2}}{2}^1\frac{2}{t} \, du = 2 (-\ln \frac{\sqrt{2}}{2})=\boxed{\ln 2}
\end{gather*}

\problem{7}

\subproblem{a}
\begin{gather*}
    x=\frac{1}{t}\\
    dx=-\frac{1}{t^2}\,dt\\
    \int_\frac{1}{a}^af(x)\frac{dx}{x}= -\int_a^\frac{1}{a}f\paren{\frac{1}{t}} \frac{dt}{t}=\int_\frac{1}{a}^af\paren{\frac{1}{x}}\frac{dx}{x}
\end{gather*}

\subproblem{b}
\begin{gather*}
    2\int_\frac{1}{2}^2\frac{\ln x}{\frac{1}{x}+x}\, \frac{dx}{x} =\int_\frac{1}{2}^2 \paren{\frac{\ln x}{\frac{1}{x}+x} + \frac{\ln \frac{1}{x}}{1+\frac{1}{x^2}}} \frac{dx}{x}=\int_\frac{1}{2}^2 \paren{\frac{\ln x - \ln x}{\frac{1}{x}+x}}\frac{dx}{x}=0\\
    \boxed{\int_\frac{1}{2}^2\frac{\ln x}{1+x^2}\, dx=0}
\end{gather*}

\subproblem{c}
\begin{gather*}
    2 \int_\frac{1}{2}^2 \arctan x\, \frac{dx}{x} = \int_\frac{1}{2}^2 \paren{\arctan x +\arctan \frac{1}{x}} \frac{dx}{x} \\= \int_\frac{1}{2}^2 \paren{\arctan x + \frac{\pi}{2}-\arctan x} \frac{dx}{x}= \frac{\pi}{2}\int_\frac{1}{2}^2 \frac{dx}{x}= \frac{\pi}{2} (\ln 2 - \ln \frac{1}{2}) = \pi \ln 2\\
    \int_\frac{1}{2}^2 \frac{\arctan x}{x} \, dx = \boxed{\frac{\pi}{2} \ln 2}
\end{gather*}

\subproblem{d}
\begin{gather*}
    \int \frac{x^2+1}{x^4+1} \, dx=\int \frac{1+\frac{1}{x^2}}{x^2+\frac{1}{x^2}}\, dx\\
    u=x-\frac{1}{x}\\
    u^2=x^2+\frac{1}{x^2}-2\\
    du=1+\frac{1}{x^2}\, dx\\
    \int \frac{1+\frac{1}{x^2}}{x^2+\frac{1}{x^2}}\, dx = \int \frac{1}{2+u^2} \, du\\
    t= \frac{u}{\sqrt{2}}\\
    dt = \frac{du}{\sqrt{2}}\\
    \int \frac{1}{2+u^2} \, du = \frac{1}{\sqrt{2}}\int \frac{1}{1+t^2}\, dt = \frac{1}{\sqrt{2}}\arctan \frac{u}{\sqrt{2}}+C = \boxed{\frac{1}{\sqrt{2}}\arctan \paren{\frac{x^2-1}{\sqrt{2x}}}+C}
\end{gather*}

\subproblem{d}
The value 0 is impossible, as the integrand is strictly greater than one on this interval. The error occurs as there is a jump discontinunity in the function's antiderivative.
\begin{gather*}
    \int_{-1}^1\frac{x^2+1}{x^4+1}\, dx=2\int_0^1 \frac{x^2+1}{x^4+1}\, dx=2 (0- \frac{\pi}{2\sqrt{2}})=\boxed{\frac{\pi}{2}}
\end{gather*}
